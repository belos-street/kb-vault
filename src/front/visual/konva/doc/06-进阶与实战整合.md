### 3.3 图形绘制（Canvas.js）

```js
function setupCanvas(state) {
  const { stage, layer, tr } = state;
  let isDrawing = false;
  let currentShape = null;
  let startPos = null;

  stage.on('mousedown', (e) => {
    if (e.target === stage) {
      tr.nodes([]);
    }

    if (state.currentTool === 'select' || e.target !== stage) return;

    isDrawing = true;
    startPos = stage.getPointerPosition();

    switch (state.currentTool) {
      case 'rect':
        currentShape = new Konva.Rect({
          x: startPos.x, y: startPos.y,
          width: 0, height: 0,
          fill: 'rgba(74, 144, 217, 0.3)',
          stroke: '#4A90D9', strokeWidth: 2,
          draggable: true, name: 'shape',
        });
        break;

      case 'circle':
        currentShape = new Konva.Ellipse({
          x: startPos.x, y: startPos.y,
          radiusX: 0, radiusY: 0,
          fill: 'rgba(231, 76, 60, 0.3)',
          stroke: '#E74C3C', strokeWidth: 2,
          draggable: true, name: 'shape',
        });
        break;

      case 'text':
        const text = prompt('请输入文字：', '文字');
        if (text) {
          currentShape = new Konva.Text({
            x: startPos.x, y: startPos.y,
            text, fontSize: 24, fill: '#333',
            draggable: true, name: 'shape',
          });
          layer.add(currentShape);
          bindShapeEvents(currentShape, state);
          saveState(state);
        }
        isDrawing = false;
        return;
    }

    if (currentShape) layer.add(currentShape);
  });

  stage.on('mousemove', () => {
    if (!isDrawing || !currentShape) return;
    const pos = stage.getPointerPosition();

    if (state.currentTool === 'rect') {
      currentShape.setAttrs({
        width: pos.x - startPos.x,
        height: pos.y - startPos.y,
      });
    } else if (state.currentTool === 'circle') {
      const rx = Math.abs(pos.x - startPos.x) / 2;
      const ry = Math.abs(pos.y - startPos.y) / 2;
      currentShape.setAttrs({
        radiusX: rx, radiusY: ry,
        x: (startPos.x + pos.x) / 2,
        y: (startPos.y + pos.y) / 2,
      });
    }
  });

  stage.on('mouseup', () => {
    if (!isDrawing || !currentShape) return;

    if (currentShape.width() < 5 && currentShape.height() < 5) {
      currentShape.destroy();
    } else {
      bindShapeEvents(currentShape, state);
      state.shapes.push(currentShape);
      saveState(state);
    }

    currentShape = null;
    isDrawing = false;
  });
}

// 自由画笔（单独的逻辑）
let isPenDown = false;
let penPoints = [];
let penLine = null;

stage.on('mousedown', (e) => {
  if (state.currentTool !== 'pen' || e.target !== stage) return;

  isPenDown = true;
  const pos = stage.getPointerPosition();
  penPoints = [pos.x, pos.y];

  penLine = new Konva.Line({
    points: penPoints,
    stroke: '#333', strokeWidth: 3,
    lineCap: 'round', lineJoin: 'round',
    tension: 0.5, draggable: true, name: 'shape',
  });

  layer.add(penLine);
});

stage.on('mousemove', () => {
  if (!isPenDown) return;
  const pos = stage.getPointerPosition();
  penPoints.push(pos.x, pos.y);
  penLine.points(penPoints);
});

stage.on('mouseup', () => {
  if (state.currentTool !== 'pen' || !isPenDown) return;

  isPenDown = false;
  bindShapeEvents(penLine, state);
  state.shapes.push(penLine);
  saveState(state);
  penLine = null;
});
```

> **注意：两个 `mousedown` 处理器的冲突避免** —— 上面有两段 `stage.on('mousedown', ...)` ，分别是图形绘制和自由画笔。它们都绑定在同一个 Stage 上，每次点击两个处理器都会触发。避免冲突的关键是**互斥的 `currentTool` 守卫**：
> - 图形绘制：`if (state.currentTool === 'select' || ...) return;` —— 选择模式直接跳过
> - 自由画笔：`if (state.currentTool !== 'pen' || ...) return;` —— 非画笔模式直接跳过
>
> 当 `currentTool` 为 `'pen'` 时，图形绘制处理器在第一个 `if` 就 return 了；反之亦然。这个模式可以安全地扩展到更多工具，只需在各自处理器入口加上对应的工具判断。

### 3.4 图形事件绑定

```js
function bindShapeEvents(shape, state) {
  const { layer, tr } = state;

  shape.on('mouseover', () => {
    document.body.style.cursor = 'pointer';
  });

  shape.on('mouseout', () => {
    document.body.style.cursor = 'default';
  });

  shape.on('click', (e) => {
    e.cancelBubble = true;
    tr.nodes([shape]);
  });

  shape.on('dragmove', () => {
    tr.nodes([shape]);
  });

  shape.on('dragend', () => {
    saveState(state);
  });

  shape.on('transformend', () => {
    const w = shape.width() * shape.scaleX();
    const h = shape.height() * shape.scaleY();
    shape.setAttrs({ width: w, height: h, scaleX: 1, scaleY: 1 });
    saveState(state);
  });
}
```

### 3.5 撤销/重做（History.js）

使用**快照模式**：每次操作后保存整个 Layer 的 JSON，撤销时恢复到上一个快照。

```js
class HistoryManager {
  constructor(state, maxSteps = 20) {
    this.state = state;
    this.maxSteps = maxSteps;
    this.undoStack = [];
    this.redoStack = [];
  }

  saveState() {
    const { layer } = this.state;
    this.undoStack.push(layer.toJSON());
    this.redoStack = [];
    if (this.undoStack.length > this.maxSteps) this.undoStack.shift();
  }

  undo() {
    if (this.undoStack.length <= 1) return;
    this.redoStack.push(this.undoStack.pop());
    this._restore(this.undoStack[this.undoStack.length - 1]);
  }

  redo() {
    if (this.redoStack.length === 0) return;
    this.undoStack.push(this.redoStack.pop());
    this._restore(this.undoStack[this.undoStack.length - 1]);
  }

  // _restore 的核心思路：
  // 1. 记录当前选中的图形 ID
  // 2. 清空 Layer 所有子元素
  // 3. 从 JSON 快照重建图形并逐个添加到 Layer（跳过 Transformer）
  // 4. 重新绑定事件（因为序列化不保留事件）
  // 5. 恢复之前的选中状态
  _restore(json) {
    const { layer, tr } = this.state;
    const selectedIds = tr.nodes().map(n => n.id());

    layer.destroyChildren();

    const tempStage = Konva.Node.create(
      `{"attrs":{},"className":"Layer","children":${JSON.stringify(json)}}`
    );
    tempStage.find('*').forEach(child => {
      if (child.getClassName() !== 'Transformer') {
        const clone = child.clone();
        layer.add(clone);
        bindShapeEvents(clone, this.state);
      }
    });

    if (selectedIds.length > 0) {
      const nodes = selectedIds
        .map(id => layer.findOne('#' + id))
        .filter(Boolean);
      tr.nodes(nodes);
    }
  }
}
```

快捷键绑定：

```js
const history = new HistoryManager(state);

document.addEventListener('keydown', (e) => {
  if (e.ctrlKey && e.key === 'z' && !e.shiftKey) {
    e.preventDefault();
    history.undo();
  }
  if ((e.ctrlKey && e.key === 'y') || (e.ctrlKey && e.shiftKey && e.key === 'z')) {
    e.preventDefault();
    history.redo();
  }
  if (e.key === 'Delete' || e.key === 'Backspace') {
    e.preventDefault();
    tr.nodes().forEach(n => n.destroy());
    tr.nodes([]);
    history.saveState();
  }
});
```

### 3.6 图层管理（LayerPanel.js）

```js
class LayerManager {
  constructor(stage) {
    this.stage = stage;
    this.layers = [];
    this.currentIndex = 0;
  }

  addLayer(name = `Layer ${this.layers.length + 1}`) {
    const layer = new Konva.Layer({ name });
    this.stage.add(layer);
    this.layers.push(layer);
    this.currentIndex = this.layers.length - 1;
    this.renderPanel();
    return layer;
  }

  removeLayer(index) {
    if (this.layers.length <= 1) return;
    this.layers[index].destroy();
    this.layers.splice(index, 1);
    this.currentIndex = Math.min(this.currentIndex, this.layers.length - 1);
    this.renderPanel();
  }

  switchLayer(index) {
    this.currentIndex = index;
    this.renderPanel();
  }

  toggleVisibility(index) {
    const layer = this.layers[index];
    layer.visible(!layer.visible());
    this.renderPanel();
  }

  moveUp(index) {
    if (index >= this.layers.length - 1) return;
    [this.layers[index], this.layers[index + 1]] =
      [this.layers[index + 1], this.layers[index]];
    this.stage.setChildren(this.layers);
    this.renderPanel();
  }

  moveDown(index) {
    if (index <= 0) return;
    [this.layers[index], this.layers[index - 1]] =
      [this.layers[index - 1], this.layers[index]];
    this.stage.setChildren(this.layers);
    this.renderPanel();
  }

  renderPanel() {
    const panel = document.getElementById('layer-panel');
    panel.innerHTML = '';

    this.layers.forEach((layer, i) => {
      const item = document.createElement('div');
      item.className = 'layer-item' + (i === this.currentIndex ? ' active' : '');

      const visibleBtn = document.createElement('button');
      visibleBtn.textContent = layer.visible() ? '👁' : '👁‍🗨';
      visibleBtn.onclick = () => this.toggleVisibility(i);

      const nameSpan = document.createElement('span');
      nameSpan.textContent = layer.name();
      nameSpan.onclick = () => this.switchLayer(i);

      item.appendChild(visibleBtn);
      item.appendChild(nameSpan);
      panel.appendChild(item);
    });
  }
}
```

function setupExport(stage) {

    link.click();

    const blob = new Blob([stage.toJSON()], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = 'project.json';
    link.click();
    URL.revokeObjectURL(url);
  };

  document.getElementById('loadProject').onclick = () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.json';
    input.onchange = (e) => {
      const reader = new FileReader();
      reader.onload = (evt) => {
        try {
          const newStage = Konva.Node.create(evt.target.result, 'container');
          newStage.find('Layer').forEach(l => l.draw());
          alert('项目已加载');
        } catch {
          alert('文件格式错误');
        }
      };
      reader.readAsText(e.target.files[0]);
    };
    input.click();
  };

  document.getElementById('clearCanvas').onclick = () => {
    if (!confirm('确定清空画布？')) return;
    stage.find('Layer').forEach(l => {
      l.destroyChildren();
    });
  };
}

## 4. 性能优化

| 场景 | 优化方式 |
|------|---------|
| 频繁更新（拖拽、画笔） | 由 `setAttrs` 触发的自动批量渲染处理，无需手动调用 `batchDraw()` |
| 静态背景 | `bgRect.listening(false)` 禁用事件检测 |
| 大量相同图形 | 使用 `clone()` 复用图形配置 |
| 删除图形 | 用 `destroy()` 代替 `remove()` 释放内存 |

## 📝 面试回答模板

> **问：Konva 的撤销/重做怎么实现？**
>
> 使用快照模式：每次操作后调用 `layer.toJSON()` 保存当前状态到 `undoStack`，撤销时从栈中取出上一个快照，用 `Konva.Node.create()` 重建 Layer 内容。重做同理，从 `redoStack` 取出恢复。这种方式的优点是实现简单、数据完整，缺点是大项目时 JSON 体积较大。如果对性能要求高，可以考虑命令模式记录操作而非全量快照。

> **问：Konva 适合什么样的项目？**
>
> 适合需要高度交互的 2D 图形编辑器类项目，比如在线白板、海报编辑器、户型图标注、流程图工具等。核心场景是需要对象化管理（选中、拖拽、变换、事件绑定）而不只是静态绘制。如果是简单的数据图表，用 ECharts 更合适；如果是 2D 游戏，用 PixiJS 性能更好。
