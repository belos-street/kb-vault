> 对应大纲：模块 9 | 预计时间：5~7 天

### 2.2 玩家飞船

```javascript
import { Sprite, Assets } from 'pixi.js';

// 加载飞船纹理
const shipTexture = await Assets.load('./assets/kenney_space-shooter-extension/PNG/Sprites X2/Ships/spaceShips_001.png');

const player = new Sprite(shipTexture);
player.anchor.set(0.5);
player.scale.set(0.5);
player.position.set(400, 520);
gameScene.addChild(player);

// 键盘状态
const keys = {};
window.addEventListener('keydown', (e) => { keys[e.code] = true; });
window.addEventListener('keyup', (e) => { keys[e.code] = false; });

// 移动速度
const PLAYER_SPEED = 5;

// 游戏循环中更新玩家位置
app.ticker.add((ticker) => {
  const dt = ticker.deltaTime;
  if (!gameScene.visible) return;  // 未开始游戏不处理输入

  if (keys['ArrowLeft'] || keys['KeyA']) {
    player.x -= PLAYER_SPEED * dt;
  }
  if (keys['ArrowRight'] || keys['KeyD']) {
    player.x += PLAYER_SPEED * dt;
  }
  if (keys['ArrowUp'] || keys['KeyW']) {
    player.y -= PLAYER_SPEED * dt;
  }
  if (keys['ArrowDown'] || keys['KeyS']) {
    player.y += PLAYER_SPEED * dt;
  }

  // 边界限制
  const halfW = player.width / 2;
  const halfH = player.height / 2;
  player.x = Math.max(halfW, Math.min(800 - halfW, player.x));
  player.y = Math.max(halfH, Math.min(600 - halfH, player.y));
});
```

### 2.3 子弹系统（对象池）

```javascript
import { Graphics } from 'pixi.js';

// 复用模块 7 的 ObjectPool 模式
class ObjectPool {
  constructor(createFn, resetFn, maxSize = 50) {
    this.createFn = createFn;
    this.resetFn = resetFn;
    this.maxSize = maxSize;
    this.pool = [];
    this.active = [];
  }

  get() {
    let obj;
    if (this.pool.length > 0) {
      obj = this.pool.pop();
    } else if (this.active.length < this.maxSize) {
      obj = this.createFn();
    } else {
      return null;
    }
    this.resetFn(obj);
    obj.visible = true;
    this.active.push(obj);
    return obj;
  }

  release(obj) {
    obj.visible = false;
    const idx = this.active.indexOf(obj);
    if (idx !== -1) {
      this.active.splice(idx, 1);
      this.pool.push(obj);
    }
  }
}

// 生成子弹纹理（用 Graphics 生成，无需图片文件）
const bulletGfx = new Graphics();
bulletGfx.rect(0, 0, 4, 12);
bulletGfx.fill(0xffd93d);
const bulletTexture = app.renderer.generateTexture(bulletGfx);

// 子弹池
const bulletPool = new ObjectPool(
  () => {
    const bullet = new Sprite(bulletTexture);
    bullet.anchor.set(0.5);
    gameScene.addChild(bullet);
    return bullet;
  },
  (bullet) => {
    bullet.x = 0;
    bullet.y = 0;
    bullet.alpha = 1;
    bullet.rotation = 0;
  },
  50
);

// 发射子弹（按空格）
let shootCooldown = 0;
const SHOOT_INTERVAL = 8; // 帧间隔

app.ticker.add((ticker) => {
  const dt = ticker.deltaTime;
  if (!gameScene.visible) return;  // 未开始游戏不更新

  shootCooldown -= dt;

  // 只在游戏进行中（gameScene 可见）时响应射击
  if (keys['Space'] && shootCooldown <= 0) {
    const bullet = bulletPool.get();
    if (bullet) {
      bullet.x = player.x;
      bullet.y = player.y - 20;
    }
    shootCooldown = SHOOT_INTERVAL;
  }

  // 更新子弹位置
  const BULLET_SPEED = 8;
  for (let i = bulletPool.active.length - 1; i >= 0; i--) {
    const bullet = bulletPool.active[i];
    bullet.y -= BULLET_SPEED * dt;
    if (bullet.y < -20) {
      bulletPool.release(bullet);
    }
  }
});
```

// 敌机纹理
const enemyTexture = await Assets.load('./assets/kenney_space-shooter-extension/PNG/Sprites X2/Meteors/spaceMeteors_001.png');

// 敌机池（复用 ObjectPool）
const enemyPool = new ObjectPool(

let spawnTimer = 0;
const SPAWN_INTERVAL = 60; // 每 60 帧生成一个

app.ticker.add((ticker) => {
  const dt = ticker.deltaTime;
  if (!gameScene.visible) return;  // 未开始游戏不生成敌机

  spawnTimer -= dt;

  if (spawnTimer <= 0) {
    const enemy = enemyPool.get();
    if (enemy) {
      enemy.x = 50 + Math.random() * 700;
      enemy.y = -40;
    }
    spawnTimer = SPAWN_INTERVAL;

  // 更新敌机位置
  const ENEMY_SPEED = 2;
  for (let i = enemyPool.active.length - 1; i >= 0; i--) {
    const enemy = enemyPool.active[i];
    enemy.y += ENEMY_SPEED * dt;
    if (enemy.y > 650) {
      enemyPool.release(enemy);
    }
  }
});

### 2.5 碰撞检测

```javascript
let score = 0;

// 圆形碰撞（平方比较，性能更好）
function circleHit(a, b) {
  const dx = a.x - b.x;
  const dy = a.y - b.y;
  const r = (a.radius || 10) + (b.radius || 10);
  return dx * dx + dy * dy < r * r;
}

app.ticker.add(() => {
  if (!gameScene.visible) return;  // 未开始游戏不检测碰撞

  // 子弹 vs 敌机
  for (let i = bulletPool.active.length - 1; i >= 0; i--) {
    const bullet = bulletPool.active[i];
    for (let j = enemyPool.active.length - 1; j >= 0; j--) {
      const enemy = enemyPool.active[j];
      if (circleHit(bullet, enemy)) {
        // 命中！
        createExplosion(enemy.x, enemy.y); // 爆炸特效
        bulletPool.release(bullet);
        enemyPool.release(enemy);
        score += 10;
        scoreText.text = `Score: ${score}`;
        break;
      }
    }
  }

  // 敌机 vs 玩家
  for (let i = enemyPool.active.length - 1; i >= 0; i--) {
    const enemy = enemyPool.active[i];
    if (circleHit(player, enemy)) {
      createExplosion(enemy.x, enemy.y);
      enemyPool.release(enemy);
      lives--;
      livesText.text = `Lives: ${lives}`;
      if (lives <= 0) {
        gameOver();
      }
    }
  }
});
```

### 2.6 爆炸特效

```javascript
function createExplosion(x, y) {
  const count = 12;
  for (let i = 0; i < count; i++) {
    const p = new Graphics();
    p.circle(0, 0, 2 + Math.random() * 3);
    p.fill(i < count / 2 ? 0xff6b6b : 0xffd93d);
    p.position.set(x, y);

    const angle = (Math.PI * 2 / count) * i + Math.random() * 0.5;
    const speed = 2 + Math.random() * 3;
    p.vx = Math.cos(angle) * speed;
    p.vy = Math.sin(angle) * speed;
    p.life = 1;

    gameScene.addChild(p);

    const update = (ticker) => {
      const dt = ticker.deltaTime;
      p.x += p.vx * dt;
      p.y += p.vy * dt;
      p.life -= 0.03 * dt;
      p.alpha = p.life;
      p.scale.set(p.life);

      if (p.life <= 0) {
        gameScene.removeChild(p);  // 从场景图移除
        p.destroy();               // 释放 GPU 资源
        app.ticker.remove(update);
      }
    };
    app.ticker.add(update);
  }
}
```

app.stage.eventMode = 'static';

  if (!touchActive) return;

## 5. 面试回答模板

> **问：做一个完整的 2D 游戏需要哪些技术？**
>
> 以飞机大战为例，需要：1）**场景图管理**——用 Container 分离开始/游戏/结束三个场景；2）**精灵与纹理**——加载飞船、敌机、子弹等图片资源；3）**游戏循环**——Ticker 驱动帧更新，用 deltaTime 保证帧率无关；4）**输入处理**——键盘事件控制移动，触摸事件适配移动端；5）**碰撞检测**——圆形碰撞（距离公式）判断子弹命中；6）**对象池**——子弹和敌机复用，避免 GC 卡顿；7）**粒子特效**——爆炸效果用 Graphics 或 ParticleContainer 实现；8）**UI**——Text 动态更新分数和生命值。

> **问：对象池在飞机大战中怎么用？**
>
> 子弹和敌机是高频创建/销毁的对象。对象池预先创建一批对象放入"池子"，需要时 `get()` 取出，用完后 `release()` 归还。这样避免了每帧 `new Sprite()` + `destroy()` 带来的内存分配和 GC 压力。池的最大容量根据实际需求设定（子弹 50、敌机 20），超过上限时不再创建新对象。

> **问：飞机大战的碰撞检测怎么做？性能怎么优化？**
>
> 用圆形碰撞检测——计算两个圆心的距离平方，判断是否小于半径之和的平方（省去 Math.sqrt）。每帧遍历所有活跃子弹，对每颗子弹遍历所有活跃敌机做检测。优化方式：1）平方比较代替开方；2）子弹和敌机数量都不大（< 100），朴素遍历够用；3）如果数量大，可以加空间网格分区。
